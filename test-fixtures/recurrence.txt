# Shared recurrence fixtures: run by Jest (src/reminders/recurrence.ts) and by

# values are hand-computed. Each line:

#   timeZone | mode | localStart | startAt | intervalMinutes | after | expectedNext

# localStart: floating local "YYYY-MM-DDTHH:MM" (daily/weekly/monthly/yearly)
# startAt / after / expectedNext: UTC instants; "-" = not used / no occurrence.
# Europe/Belgrade 2026: CET (+1) → CEST (+2) on 29 Mar 02:00, back on 25 Oct 03:00.
# America/New_York 2026: EST (-5) → EDT (-4) on 8 Mar 02:00, back on 1 Nov 02:00.
